# /usr/local/bin/ssh-check

#!/bin/bash

# ============================================================
# 脚本名称: ssh-check (缓存读取版)
# 功能: 读取 /var/cache/ssh-monitor.data 并格式化输出
# 速度: 极快 (0.01s)
# ============================================================

CACHE_FILE="/var/cache/ssh-monitor.data"
MODE="normal"

if [ "$1" == "raw" ]; then
    MODE="raw"
fi

# --- 颜色定义 ---
GREEN='\033[1;32m'
YELLOW='\033[1;33m'
RED='\033[1;31m'
GREY='\033[1;30m'
CYAN='\033[1;36m'
RESET='\033[0m'

# --- 检查缓存文件是否存在 ---
if [ ! -f "$CACHE_FILE" ]; then
    echo "系统初始化中，请稍后..."
    exit 0
fi

# --- 打印表头 (仅 Normal 模式) ---
if [ "$MODE" == "normal" ]; then
    echo ""
    printf "${CYAN}%-15s %-15s %-12s %-10s${RESET}\n" "HOSTNAME" "IP ADDRESS" "SESSIONS" "STATUS"
    echo "--------------------------------------------------------"
fi

# --- 读取并格式化 ---
while read -r host ip count status; do
    
    if [ "$MODE" == "raw" ]; then
        # Raw 模式：直接原样输出，供 Web API 使用
        echo "$host $ip $count $status"
    else
        # Normal 模式：添加颜色，供 CLI 使用
        COLOR=$GREEN
        case "$status" in
            "BUSY")   COLOR=$RED ;;
            "ACTIVE") COLOR=$YELLOW ;;
            "OFFLINE") COLOR=$GREY ;;
        esac
        
        printf "%-15s %-15s ${COLOR}%-12s %-10s${RESET}\n" "$host" "$ip" "$count" "$status"
    fi

done < "$CACHE_FILE"

if [ "$MODE" == "normal" ]; then
    echo "--------------------------------------------------------"
    # 可选：显示缓存更新时间
    update_time=$(date -r "$CACHE_FILE" "+%H:%M:%S")
    echo "Last Updated: $update_time"
    echo ""
fi

**创建节点检测文件**

```bash
# /usr/local/bin/ssh-scan

#!/bin/bash

# ============================================================
# 脚本名称: ssh-scan (稳定版)
# 功能: 修复 wait 失效问题，确保目录存在直至所有任务完成
# ============================================================

CACHE_FILE="/var/cache/ssh-monitor.data"
WORK_DIR="/tmp/ssh_scan_work"
LIST_FILE="/tmp/ssh_scan_list"
HOST_KEYWORD="node"

# 1. 初始化环境
rm -rf "$WORK_DIR"
mkdir -p "$WORK_DIR"

# 2. 准备主机列表 (先写入临时文件，避免管道子Shell问题)
grep -v "^#" /etc/hosts | grep "$HOST_KEYWORD" | awk '{print $1, $2}' > "$LIST_FILE"

# 检查列表是否为空
if [ ! -s "$LIST_FILE" ]; then
    echo "错误: 未找到节点列表"
    exit 1
fi

echo "开始扫描 $(wc -l < $LIST_FILE) 个节点..."

# 3. 读取列表并启动后台任务
# 注意：这里使用 < 重定向，而不是管道 |，确保在主 Shell 中运行
while read -r ip host; do
    if [ -z "$host" ]; then continue; fi

    # 启动后台子进程
    (
        # 默认值
        final_status="OFFLINE"
        final_count="--"
        
        # 执行 SSH (强行限制 3 秒超时)
        raw=$(timeout 3 ssh -q -o BatchMode=yes -o StrictHostKeyChecking=no -o ConnectTimeout=3 root@"$host" "ss -Htn state established '( sport = :22 )' | wc -l" 2>/dev/null)
        
        # 结果判断
        if [ $? -eq 0 ] && [ -n "$raw" ]; then
            count=$((raw - 1))
            if [ "$count" -lt 0 ]; then count=0; fi
            
            final_count=$count
            
            if [ "$count" -ge 2 ]; then final_status="BUSY"
            elif [ "$count" -eq 1 ]; then final_status="ACTIVE"
            else final_status="IDLE"
            fi
        fi
        
        # 写入结果 (确保目录还存在)
        if [ -d "$WORK_DIR" ]; then
            echo "$host $ip $final_count $final_status" > "$WORK_DIR/$host"
        fi
        
    ) & 
    
done < "$LIST_FILE"

# 4. 关键：等待所有后台子进程结束
wait

# 5. 合并结果
if [ "$(ls -A $WORK_DIR 2>/dev/null)" ]; then
    cat "$WORK_DIR"/* | sort > "$CACHE_FILE"
    chmod 644 "$CACHE_FILE"
    echo "扫描完成。"
else
    echo "警告: 未生成任何数据，保留旧缓存。"
fi

# 6. 清理
rm -rf "$WORK_DIR"
rm -f "$LIST_FILE"
```

        fi

        # 2. 配置 SSH 存活检测 (针对: 异常断开的僵尸连接)
        # 修改 sshd_config，如果已有配置则修改，没有则追加
        sed -i "/ClientAliveInterval/d" /etc/ssh/sshd_config
        sed -i "/ClientAliveCountMax/d" /etc/ssh/sshd_config
        echo "ClientAliveInterval 60" >> /etc/ssh/sshd_config
        echo "ClientAliveCountMax 5" >> /etc/ssh/sshd_config

        # 3. 重启 SSHD 服务使配置生效
        systemctl restart sshd
    '

// 解析 ssh-check 输出的工具函数
function parseSSHOutput(output) {
    const lines = output.split('\n');
    const nodes = [];
    
    // 跳过表头和分割线 (根据你的脚本输出调整)
    // 假设输出格式为:
    // HOSTNAME        IP ADDRESS      SESSIONS    STATUS
    // --------------------------------------------------
    // node1           1.2.3.4         0           IDLE
    
    lines.forEach(line => {
        // 过滤空行和分隔线
        if (!line || line.startsWith('HOSTNAME') || line.startsWith('---')) return;
        
        // 使用正则匹配空白字符分隔的列
        const parts = line.trim().split(/\s+/);
        
        if (parts.length >= 4) {
            nodes.push({
                hostname: parts[0],
                ip: parts[1],
                sessions: parts[2],
                status: parts[3]
            });
        }
    });
    
    return nodes;
}

app.get('/api/status', (req, res) => {
    const conn = new Client();
    let buffer = '';

    conn.on('ready', () => {
        // 执行 guest 允许的 ssh-check 命令
        conn.exec('ssh-check', (err, stream) => {
            if (err) {
                res.status(500).json({ error: 'Failed to execute command' });
                conn.end();
                return;
            }

            stream.on('close', (code, signal) => {
                conn.end();
                // 解析数据并返回
                const parsedData = parseSSHOutput(buffer);
                res.json(parsedData);
            }).on('data', (data) => {
                buffer += data;
            }).stderr.on('data', (data) => {
                console.error('STDERR: ' + data);
            });
        });
    }).on('error', (err) => {
        console.error('SSH Connection Error:', err);
        res.status(500).json({ error: 'SSH Connection failed', details: err.message });
    }).connect(sshConfig);
});

app.listen(PORT, () => {
    console.log(`Monitoring Backend running on http://localhost:${PORT}`);

## 阶段五 Systemd 接管

描述

```bash
# 我们使用了 缓存-读取-计算-缓存 的方式处理 ssh-check 命令，但 计划任务 太慢，所以我们需要用 systemd 接管这项服务。
```

**创建服务文件**

```bash
vim /etc/systemd/system/ssh-scan.service
```

```ini
[Unit]
Description=SSH Cluster Scanner Service

[Service]
# 使用 oneshot 类型，表示运行完就退出，等待 Timer 再次唤醒
Type=oneshot
ExecStart=/usr/local/bin/ssh-scan
# 任务超时时间 (如果 10秒 没跑完强制杀掉，防止卡死)
TimeoutStartSec=10
```

**创建定时器文件**

```bash
vim /etc/systemd/system/ssh-scan.timer
```

```ini
[Unit]
Description=Run ssh-scan every 5 seconds

[Timer]
# 开机 1 分钟后开始运行
OnBootSec=1min

# 核心设置：上一次任务结束后，等待 5 秒再次运行
# 这能有效防止“雪崩效应” (如果网络卡了，不会堆积几百个进程)
OnUnitActiveSec=5s

# 提高精度 (Systemd 默认有 1分钟的时间抖动，设为 1s 强制精确执行)
AccuracySec=1s

[Install]
WantedBy=timers.target
```

**重启 Systemd**

```bash
systemctl daemon-reload
```

---
- name: 部署远程 SSH 监控 (日志回传控制节点)
  hosts: dev
  vars:
    server_ip: "192.168.40.128"
  tasks:
    
    # 1. 配置 /etc/ssh/sshrc
    # 使用 logger 命令通过 UDP (-n) 发送日志
    - name: 配置 SSH 登录钩子 (远程发送模式)
      copy:
        dest: /etc/ssh/sshrc
        mode: '0755'
        content: |
          #!/bin/bash
          # ---------------------------------------------------
          # SSH 远程监控钩子
          # ---------------------------------------------------
          
          # 控制节点 IP (由 Ansible 变量注入)
          SERVER_IP="{{ server_ip }}"
          
          # 获取当前时间 (ISO 8601格式)
          NOW=$(date '+%Y-%m-%d %H:%M:%S')
          
          # 获取本机主机名 (方便老师区分是哪台机器连的)
          MY_HOSTNAME=$(hostname)
          
          # 获取来源 IP
          CLIENT_IP=$(echo $SSH_CONNECTION | awk '{print $1}')
          
          # 构建日志内容
          MSG="Node: $MY_HOSTNAME | User: $USER | From: $CLIENT_IP | Time: $NOW"
          
          # 使用 logger 发送 UDP 数据包到控制节点
          # -n: 指定远程服务器
          # -P: 指定端口 (默认514)
          # -t: 指定 Tag (我们在控制节点通过这个 Tag SSH_MONITOR 来识别)
          # -d: 使用 UDP (部分老版本 logger 需要，新版自动识别，为兼容性不加-d，默认UDP或TCP取决于配置，通常 -n 隐含 UDP)
          # 为了最大兼容性 RHEL9，使用如下格式：
          
          logger -d -n "$SERVER_IP" -P 514 -t "SSH_MONITOR" "$MSG" 2>/dev/null

    # 2. 验证连通性 (可选，发送一条测试日志)
    - name: 发送一条测试日志
      shell: logger -n "{{ server_ip }}" -P 514 -t "SSH_MONITOR" "Ansible_Deployment_Test_Connection"
      ignore_errors: yes
